students = []


def add_student():
    print("\n===== ADD STUDENT =====")

    student_id = input("Enter Student ID: ")

    for student in students:
        if student["id"] == student_id:
            print("Error: Student ID already exists.")
            return

    name = input("Enter Student Name: ")

    # ADDED: Sex Selection
    print("\nSelect Sex:")
    print("1. Male")
    print("2. Female")

    sex_choice = input("Enter your choice: ")

    if sex_choice == "1":
        sex = "Male"
    elif sex_choice == "2":
        sex = "Female"
    else:
        print("Invalid sex choice.")
        return

    course = input("Enter Course: ")
    year = input("Enter Year Level: ")

    student = {
        "id": student_id,
        "name": name,
        "sex": sex,     
        "course": course,
        "year": year,
        "status": "Active"      
    }

    students.append(student)

    print("Student added successfully!")
    
def view_students():
    print("\n===== ACTIVE STUDENT LIST =====")

    if len(students) == 0:
        print("No students found.")
        return

    print("-" * 85)
    print(f"{'ID':<15}{'NAME':<25}{'SEX':<12}{'COURSE':<15}{'YEAR':<10}")
    print("-" * 85)

    for student in students:

        # ADDED: Show only active students
        if student["status"] == "Active":

            print(
                f"{student['id']:<15}"
                f"{student['name']:<25}"
                f"{student['sex']:<12}"
                f"{student['course']:<15}"
                f"{student['year']:<10}"
            )

    print("-" * 85)

def search_student():
    print("\n===== SEARCH STUDENT =====")

    student_id = input("Enter Student ID: ")

    for student in students:
        if student["id"] == student_id and student["status"] == "Active":
            print("\nStudent Found!")
            print("Student ID :", student["id"])
            print("Name       :", student["name"])
            print("Sex        :", student["sex"])
            print("Course     :", student["course"])
            print("Year Level :", student["year"])
            return

    print("Student not found.")

def update_student():
    print("\n===== UPDATE STUDENT =====")

    student_id = input("Enter Student ID to update: ")

    for student in students:
        if student["id"] == student_id and student["status"] == "Active":

            print("\nCurrent Information:")
            print("Name       :", student["name"])
            print("Sex        :", student["sex"])    
            print("Course     :", student["course"])
            print("Year Level :", student["year"])

            print("\nEnter new information.")
            print("Press ENTER to keep the current value.")

            name = input("New Name: ")

            # ADDED: Update Sex
            print("\nNew Sex:")
            print("1. Male")
            print("2. Female")
            print("3. Keep Current Sex")

            sex_choice = input("Enter your choice: ")

            if sex_choice == "1":
                student["sex"] = "Male"
            elif sex_choice == "2":
                student["sex"] = "Female"
            elif sex_choice == "3" or sex_choice == "":
                pass
            else:
                print("Invalid sex choice. Current sex will remain.")

            course = input("New Course: ")
            year = input("New Year Level: ")

            if name != "":
                student["name"] = name

            if course != "":
                student["course"] = course

            if year != "":
                student["year"] = year

            print("\nStudent updated successfully!")
            return

    print("Student not found.")

def delete_student():
    print("\n===== DELETE STUDENT =====")

    student_id = input("Enter Student ID to delete: ")

    for student in students:
        if student["id"] == student_id and student["status"] == "Active":

            print("\nStudent Found:")
            print("Name:", student["name"])

            confirm = input("Are you sure you want to delete? (Y/N): ")

            if confirm.upper() == "Y":
         
                student["status"] = "Deleted"

                print("Student deleted successfully!")
            else:
                print("Delete operation cancelled.")

            return

    print("Student not found.")

def view_deleted_students():
    print("\n===== DELETED STUDENT LIST =====")

    deleted_found = False

    print("-" * 85)
    print(f"{'ID':<15}{'NAME':<25}{'SEX':<12}{'COURSE':<15}{'YEAR':<10}")
    print("-" * 85)

    for student in students:

        if student["status"] == "Deleted":

            deleted_found = True

            print(
                f"{student['id']:<15}"
                f"{student['name']:<25}"
                f"{student['sex']:<12}"
                f"{student['course']:<15}"
                f"{student['year']:<10}"
            )

    print("-" * 85)

    if deleted_found == False:
        print("No deleted students found.")

def restore_student():
    print("\n===== RESTORE STUDENT =====")

    student_id = input("Enter Student ID to restore: ")

    for student in students:

        if student["id"] == student_id and student["status"] == "Deleted":

            print("\nDeleted Student Found:")
            print("Student ID :", student["id"])
            print("Name       :", student["name"])
            print("Sex        :", student["sex"])
            print("Course     :", student["course"])
            print("Year Level :", student["year"])

            confirm = input("\nRestore this student? (Y/N): ")

            if confirm.upper() == "Y":

                student["status"] = "Active"

                print("Student restored successfully!")

            else:
                print("Restore operation cancelled.")

            return

    print("Deleted student not found.")
